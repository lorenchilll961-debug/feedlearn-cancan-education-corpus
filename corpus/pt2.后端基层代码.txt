def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in app.config['ALLOWED_EXTENSIONS']

def extract_text_from_pdf(file_path):
    text = ""
    try:
        with open(file_path, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            for page in pdf_reader.pages:
                text += page.extract_text() + "\n"
    except Exception as e:
        print(f"PDF读取错误: {e}")
    return text

def extract_text_from_docx(file_path):
    text = ""
    try:
        doc = docx.Document(file_path)
        for para in doc.paragraphs:
            text += para.text + "\n"
    except Exception as e:
        print(f"Word文档读取错误: {e}")
    return text

def extract_text_from_txt(file_path):
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            return file.read()
    except:
        with open(file_path, 'r', encoding='gbk') as file:
            return file.read()

def split_text_into_chunks(text, chunk_size=500):
    words = text.split()
    chunks = []
    current_chunk = []
    current_length = 0
    
    for word in words:
        current_chunk.append(word)
        current_length += 1
        
        if current_length >= chunk_size:
            chunks.append(' '.join(current_chunk))
            current_chunk = []
            current_length = 0
    
    if current_chunk:
        chunks.append(' '.join(current_chunk))
    
    return chunks

def extract_keywords(text):
    words = text.split()
    return list(set([w for w in words if len(w) > 3]))[:10]

def assess_difficulty(text):
    avg_word_length = np.mean([len(w) for w in text.split()])
    if avg_word_length < 5:
        return "初级"
    elif avg_word_length < 7:
        return "中级"
    else:
        return "高级"

def build_knowledge_index():
    if not knowledge_base['documents']:
        return
    
    print("🔨 灿灿正在构建知识索引...")
    texts = [doc['text'] for doc in knowledge_base['documents']]
    embeddings = embedding_model.encode(texts)
    
    dimension = embeddings.shape[1]
    index = faiss.IndexFlatL2(dimension)
    index.add(embeddings.astype('float32'))
    
    knowledge_base['embeddings'] = embeddings
    knowledge_base['index'] = index
    print(f"✨ 知识库构建完成！共 {len(texts)} 个知识片段")

def search_knowledge_base(query, top_k=3):
    if knowledge_base['index'] is None:
        return []
    
    query_embedding = embedding_model.encode([query])
    distances, indices = knowledge_base['index'].search(query_embedding.astype('float32'), top_k)
    
    results = []
    for idx, distance in zip(indices[0], distances[0]):
        if idx < len(knowledge_base['documents']):
            results.append({
                'text': knowledge_base['documents'][idx]['text'],
                'source': knowledge_base['documents'][idx]['source'],
                'score': float(distance),
                'keywords': knowledge_base['documents'][idx].get('keywords', [])
            })
    
    return results

@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({'success': False, 'message': '没有文件被上传'})
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({'success': False, 'message': '没有选择文件'})
    
    if file and allowed_file(file.filename):
        filename = secure_filename(file.filename)
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        filename = f"{timestamp}_{filename}"
        filepath = os.path.join(app.config['UPLOAD_FOLDER'], filename)
        file.save(filepath)
        
        ext = filename.rsplit('.', 1)[1].lower()
        if ext == 'pdf':
            text = extract_text_from_pdf(filepath)
        elif ext == 'docx':
            text = extract_text_from_docx(filepath)
        else:
            text = extract_text_from_txt(filepath)
        
        if not text.strip():
            return jsonify({'success': False, 'message': '无法从文件中提取文本内容'})
        
        chunks = split_text_into_chunks(text)
        keywords = extract_keywords(text)
        difficulty = assess_difficulty(text)
        
        for chunk in chunks:
            knowledge_base['documents'].append({
                'text': chunk,
                'source': filename,
                'upload_time': timestamp,
                'keywords': keywords,
                'difficulty': difficulty
            })
        
        build_knowledge_index()
        
        return jsonify({
            'success': True, 
            'message': f'文件上传成功！已添加 {len(chunks)} 个知识片段',
            'filename': filename,
            'chunks': len(chunks),
            'difficulty': difficulty,
            'keywords': keywords[:5]
        })
    
    return jsonify({'success': False, 'message': '不支持的文件格式'})

@app.route('/ask', methods=['POST'])
def ask_question():
    data = request.json
    question = data.get('question', '')
    style = data.get('style', 'professional')
    language = data.get('language', 'zh')
    
    if not question:
        return jsonify({'success': False, 'message': '问题不能为空'})
    
    if not knowledge_base['documents']:
        return jsonify({'success': False, 'message': '知识库为空，请先上传文档'})
    
    relevant_docs = search_knowledge_base(question, top_k=3)
    context = "\n\n".join([doc['text'] for doc in relevant_docs])
    answer = generate_answer_with_cancan(question, context, style, language)
    
    return jsonify({
        'success': True,
        'answer': answer,
        'sources': [{'source': doc['source'], 'score': doc['score']} for doc in relevant_docs]
    })

@app.route('/extract_keywords', methods=['POST'])
def keywords():
    if not knowledge_base['documents']:
        return jsonify({'success': False, 'message': '知识库为空'})
    
    all_keywords = []
    for doc in knowledge_base['documents']:
        all_keywords.extend(doc.get('keywords', []))
    
    keyword_freq = {}
    for kw in all_keywords:
        keyword_freq[kw] = keyword_freq.get(kw, 0) + 1
    
    sorted_keywords = sorted(keyword_freq.items(), key=lambda x: x[1], reverse=True)[:20]
    
    return jsonify({
        'success': True, 
        'keywords': [{'word': k, 'freq': v} for k, v in sorted_keywords]
    })

@app.route('/stats')
def get_stats():
    difficulty_stats = {}
    for doc in knowledge_base['documents']:
        diff = doc.get('difficulty', '未知')
        difficulty_stats[diff] = difficulty_stats.get(diff, 0) + 1

        'total_documents': len(knowledge_base['documents']),
        'total_files': len(set([doc['source'] for doc in knowledge_base['documents']])),
        'difficulty_distribution': difficulty_stats
    })
