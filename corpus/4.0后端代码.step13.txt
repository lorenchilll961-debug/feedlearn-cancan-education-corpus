def allowed_file(filename):
    """检查文件扩展名是否允许"""
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in app.config['ALLOWED_EXTENSIONS']

def load_documents_db():
    """加载文档数据库"""
    try:
        if os.path.exists(app.config['DOCUMENTS_DB']):
            with open(app.config['DOCUMENTS_DB'], 'r', encoding='utf-8') as f:
                return json.load(f)
    except Exception as e:
        logger.error(f"加载文档数据库失败: {e}")
    return []

def save_documents_db(documents):
    """保存文档数据库"""
    try:
        with open(app.config['DOCUMENTS_DB'], 'w', encoding='utf-8') as f:
            json.dump(documents, f, ensure_ascii=False, indent=2)
        return True
    except Exception as e:
        logger.error(f"保存文档数据库失败: {e}")
        return False

def get_document_stats():
    """获取文档统计信息"""
    documents = load_documents_db()
    return {
        'total': len(documents),
        'pdf': len([d for d in documents if d.get('filename', '').endswith('.pdf')]),
        'docx': len([d for d in documents if d.get('filename', '').endswith('.docx')]),
        'txt': len([d for d in documents if d.get('filename', '').endswith('.txt')]),
        'total_chunks': sum(d.get('chunks', 0) for d in documents)
    }

@app.route('/upload', methods=['POST'])
def upload_file():
    """上传文件"""
    try:
        if 'files' not in request.files:
            return jsonify({'status': 'error', 'message': '没有文件'}), 400
        
        files = request.files.getlist('files')
        if not files:
            return jsonify({'status': 'error', 'message': '没有选择文件'}), 400
        
        uploaded_files = []
        total_chunks = 0
        
        for file in files:
            if file and allowed_file(file.filename):
                # 保存文件
                filename = secure_filename(file.filename)
                file_id = str(uuid.uuid4())
                file_ext = filename.rsplit('.', 1)[1].lower()
                saved_filename = f"{file_id}.{file_ext}"
                filepath = os.path.join(app.config['UPLOAD_FOLDER'], saved_filename)
                file.save(filepath)
                
                logger.info(f"文件已保存: {filename} -> {saved_filename}")
                
                # 处理文档
                try:
                    text_content = doc_processor.extract_text(filepath, file_ext)
                    
                    if not text_content.strip():
                        logger.warning(f"文件内容为空: {filename}")
                        os.remove(filepath)
                        continue
                    
                    # 分块
                    chunks = doc_processor.chunk_text(text_content)
                    
                    # 生成摘要
                    summary = summarizer.generate_summary(text_content[:2000])
                    
                    # 提取关键词
                    keywords = doc_processor.extract_keywords(text_content)
                    
                    # 添加到向量数据库
                    metadatas = [{
                        'source': filename,
                        'file_id': file_id,
                        'chunk_index': i,
                        'timestamp': datetime.now().isoformat()
                    } for i in range(len(chunks))]
                    
                    vector_store.add_texts(chunks, metadatas)
                    
                    # 保存文档信息
                    doc_info = {
                        'id': file_id,
                        'filename': filename,
                        'filepath': filepath,
                        'size': os.path.getsize(filepath),
                        'chunks': len(chunks),
                        'summary': summary,
                        'keywords': keywords,
                        'uploaded': datetime.now().isoformat(),
                        'type': file_ext
                    }
                    
                    documents = load_documents_db()
                    documents.append(doc_info)
                    save_documents_db(documents)
                    
                    uploaded_files.append(doc_info)
                    total_chunks += len(chunks)
                    
                    logger.info(f"文档处理成功: {filename}, 生成 {len(chunks)} 个chunks")
                    
                except Exception as e:
                    logger.error(f"处理文档失败 {filename}: {e}")
                    if os.path.exists(filepath):
                        os.remove(filepath)
                    continue
        
        if not uploaded_files:
            return jsonify({
                'status': 'error',
                'message': '没有成功上传的文件'
            }), 400
        
        return jsonify({
            'status': 'success',
            'message': f'成功上传 {len(uploaded_files)} 个文件',
            'files': uploaded_files,
            'chunks': total_chunks
        })
        
    except Exception as e:
        logger.error(f"上传失败: {e}")
        return jsonify({
            'status': 'error',
            'message': f'上传失败: {str(e)}'
        }), 500

@app.route('/get_documents', methods=['GET'])
def get_documents():
    """获取所有文档"""
    try:
        documents = load_documents_db()
        stats = get_document_stats()
        
        # 提取所有关键词
        all_keywords = []
        for doc in documents:
            if 'keywords' in doc:
                all_keywords.extend(doc['keywords'])
        
        # 统计关键词频率
        keyword_freq = {}
        for kw in all_keywords:
            keyword_freq[kw] = keyword_freq.get(kw, 0) + 1
        
        keywords = [{'word': k, 'weight': v} for k, v in keyword_freq.items()]
        
        return jsonify({
            'status': 'success',
            'documents': documents,
            'stats': stats,
            'keywords': keywords
        })
    except Exception as e:
        logger.error(f"获取文档失败: {e}")
        return jsonify({
            'status': 'error',
            'message': str(e)
        }), 500

@app.route('/chat', methods=['POST'])
def chat():
    """聊天接口"""
    try:
        data = request.get_json()
        question = data.get('question', '').strip()
        language = data.get('language', 'zh-CN')
        style = data.get('style', 'professional')
        difficulty = data.get('difficulty', 'medium')
        
        if not question:
            return jsonify({
                'status': 'error',
                'message': '问题不能为空'
            }), 400
        
        logger.info(f"收到问题: {question}")
        
        # 检索相关文档
        results = vector_store.similarity_search(question, k=3)
        
        if not results:
            return jsonify({
                'status': 'success',
                'answer': '抱歉，我在已上传的文档中没有找到相关信息。请确保已上传相关文档。',
                'sources': []
            })
        
        # 构建上下文
        context = "\n\n".join([doc['content'] for doc in results])
        sources = [{'source': doc['metadata'].get('source', ''), 
                   'chunk': doc['metadata'].get('chunk_index', 0)} 
                  for doc in results]
        
        # 生成回答
        answer = llm_client.generate_answer(
            question=question,
            context=context,
            language=language,
            style=style,
            difficulty=difficulty
        )
        
        logger.info(f"生成答案完成")
        
        return jsonify({
            'status': 'success',
            'answer': answer,
            'sources': sources
        })
        
    except Exception as e:
        logger.error(f"聊天失败: {e}")
        return jsonify({
            'status': 'error',
            'message': f'处理失败: {str(e)}'
        }), 500

@app.route('/submit_answers', methods=['POST'])
def submit_answers():
    """提交答案并批改"""
    try:
        data = request.get_json()
        questions = data.get('questions', [])
        answers = data.get('answers', [])
        
        if len(questions) != len(answers):
            return jsonify({
                'status': 'error',
                'message': '题目和答案数量不匹配'
            }), 400
        
        results = []
        
        for i, (question, user_answer) in enumerate(zip(questions, answers)):
            q_type = question.get('type', 'multiple_choice')
            correct_answer = question.get('answer')
            
            is_correct = False
            
            if q_type == 'multiple_choice':
                # 选择题
                is_correct = str(user_answer) == str(correct_answer)
            elif q_type == 'true_false':
                # 判断题
                is_correct = str(user_answer).lower() == str(correct_answer).lower()
            else:
                # 简答题 - 使用LLM评分
                is_correct = llm_client.evaluate_answer(
                    question=question['question'],
                    user_answer=user_answer,
                    reference_answer=correct_answer
                )
            
            results.append({
                'questionIndex': i,
                'correct': is_correct,
                'userAnswer': user_answer,
                'correctAnswer': correct_answer,
                'explanation': question.get('explanation', '')
            })
        
        return jsonify({
            'status': 'success',
            'results': results
        })
        
    except Exception as e:
        logger.error(f"批改答案失败: {e}")
        return jsonify({
            'status': 'error',
            'message': str(e)
        }), 500

@app.route('/analyze_document', methods=['POST'])

        # 分析文档
        analysis = {
            'readability': doc_processor.calculate_readability(text),
            'complexity': doc_processor.assess_complexity(text),
            'topics': doc_processor.extract_topics(text),
            'word_count': len(text.split()),
            'sentence_count': len([s for s in text.split('.') if s.strip()])
        }
        
        return jsonify({
            'status': 'success',
            'analysis': analysis
        })
        
    except Exception as e:
        logger.error(f"分析文档失败: {e}")
        return jsonify({
            'status': 'error',
            'message': str(e)
