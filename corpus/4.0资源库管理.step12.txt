/* ============================================

class ResourceManager {
    constructor() {
        this.resources = [];
        this.filteredResources = [];
        this.currentSort = 'date';
        this.currentDifficulty = 'all';
        
        this.initElements();
        this.initEventListeners();
        this.loadResources();
    }
    
    initElements() {
        this.resourceGrid = document.getElementById('resourceGrid');
        this.searchInput = document.getElementById('searchInput');
        this.difficultyFilter = document.getElementById('difficultyFilter');
        this.sortBy = document.getElementById('sortBy');
        this.refreshBtn = document.getElementById('refreshBtn');
        
        // 统计元素
        this.totalDocs = document.getElementById('totalDocs');
        this.totalCategories = document.getElementById('totalCategories');
        this.totalChunks = document.getElementById('totalChunks');
        
        // 分类统计
        this.pdfCount = document.getElementById('pdfCount');
        this.docxCount = document.getElementById('docxCount');
        this.txtCount = document.getElementById('txtCount');
        
        // 关键词云
        this.keywordCloud = document.getElementById('keywordCloud');
    }
    
    initEventListeners() {
        if (this.searchInput) {
            this.searchInput.addEventListener('input', Utils.debounce((e) => {
                this.filterResources();
            }, 300));
        }
        
        if (this.difficultyFilter) {
            this.difficultyFilter.addEventListener('change', (e) => {
                this.currentDifficulty = e.target.value;
                this.filterResources();
            });
        }
        
        if (this.sortBy) {
            this.sortBy.addEventListener('change', (e) => {
                this.currentSort = e.target.value;
                this.sortResources();
            });
        }
        
        if (this.refreshBtn) {
            this.refreshBtn.addEventListener('click', () => this.refreshResources());
        }
    }
    
    // === 加载资源 ===
    async loadResources() {
        try {
            const response = await fetch('/get_documents');
            const data = await response.json();
            
            if (data.status === 'success') {
                this.resources = data.documents || [];
                this.filteredResources = [...this.resources];
                this.renderResources();
                this.updateStats();
                this.renderKeywordCloud(data.keywords || []);
            }
        } catch (error) {
            console.error('Load resources error:', error);
        }
    }
    
    async refreshResources() {
        LoadingManager.show(this.resourceGrid, '正在刷新资源...');
        await this.loadResources();
        LoadingManager.hide(this.resourceGrid);
        Utils.showToast('资源已刷新', 'success');
    }
    
    // === 渲染资源 ===
    renderResources() {
        if (!this.resourceGrid) return;
        
        if (this.filteredResources.length === 0) {
            this.showEmptyState();
            return;
        }
        
        this.resourceGrid.innerHTML = '';
        
        this.filteredResources.forEach(resource => {
            const card = this.createResourceCard(resource);
            this.resourceGrid.appendChild(card);
        });
    }
    
    createResourceCard(resource) {
        const card = document.createElement('div');
        card.className = 'card glass-effect resource-card';
        card.style.animation = 'fadeIn 0.4s ease-out';
        
        const fileExt = resource.filename.split('.').pop().toLowerCase();
        const icon = FileHandler.getFileIcon(resource.filename);
        const color = this.getFileColor(fileExt);
        
        card.innerHTML = `
            <div style="padding: 24px;">
                <div style="display: flex; align-items: flex-start; gap: 16px; margin-bottom: 16px;">
                    <div style="font-size: 3rem; background: ${color}; width: 60px; height: 60px; border-radius: 12px; display: flex; align-items: center; justify-content: center; box-shadow: var(--shadow-md);">
                        ${icon}
                    </div>
                    <div style="flex: 1; min-width: 0;">
                        <h3 style="font-size: 1.1rem; margin-bottom: 8px; color: var(--gray-800); overflow: hidden; text-overflow: ellipsis; white-space: nowrap;" title="${resource.filename}">
                            ${resource.filename}
                        </h3>
                        <div style="display: flex; gap: 8px; flex-wrap: wrap; margin-bottom: 8px;">
                            <span class="badge badge-primary">${fileExt.toUpperCase()}</span>
                            ${resource.difficulty ? `<span class="badge badge-warning">${resource.difficulty}</span>` : ''}
                            ${resource.chunks ? `<span class="badge badge-success">${resource.chunks} 块</span>` : ''}
                        </div>
                        <div style="font-size: 0.85rem; color: var(--gray-500);">
                            ${resource.size ? Utils.formatFileSize(resource.size) : ''} · 
                            ${resource.uploaded ? Utils.formatTime(resource.uploaded) : ''}
                        </div>
                    </div>
                </div>
                
                ${resource.summary ? `
                    <p style="font-size: 0.9rem; color: var(--gray-600); line-height: 1.6; margin-bottom: 16px; display: -webkit-box; -webkit-line-clamp: 3; -webkit-box-orient: vertical; overflow: hidden;">
                        ${resource.summary}
                    </p>
                ` : ''}
                
                ${resource.keywords && resource.keywords.length > 0 ? `
                    <div style="display: flex; gap: 6px; flex-wrap: wrap; margin-bottom: 16px;">
                        ${resource.keywords.slice(0, 5).map(kw => `
                            <span style="padding: 4px 10px; background: var(--cyan-lighter); color: var(--klein-blue); border-radius: 12px; font-size: 0.8rem;">
                                ${kw}
                            </span>
                        `).join('')}
                    </div>
                ` : ''}
                
                <div style="display: flex; gap: 8px; padding-top: 16px; border-top: 1px solid var(--gray-200);">
                    <button class="btn-icon" onclick="resourceManager.viewResource('${resource.id}')" title="查看详情">
                        <svg width="16" height="16" viewBox="0 0 24 24" fill="currentColor">
                            <path d="M12 4.5C7 4.5 2.73 7.61 1 12c1.73 4.39 6 7.5 11 7.5s9.27-3.11 11-7.5c-1.73-4.39-6-7.5-11-7.5zM12 17c-2.76 0-5-2.24-5-5s2.24-5 5-5 5 2.24 5 5-2.24 5-5 5zm0-8c-1.66 0-3 1.34-3 3s1.34 3 3 3 3-1.34 3-3-1.34-3-3-3z"/>
                        </svg>
                    </button>
                    <button class="btn-icon" onclick="resourceManager.analyzeResource('${resource.id}')" title="深度分析">
                        <svg width="16" height="16" viewBox="0 0 24 24" fill="currentColor">
                            <path d="M19 3H5c-1.1 0-2 .9-2 2v14c0 1.1.9 2 2 2h14c1.1 0 2-.9 2-2V5c0-1.1-.9-2-2-2zM9 17H7v-7h2v7zm4 0h-2V7h2v10zm4 0h-2v-4h2v4z"/>
                        </svg>
                    </button>
                    <button class="btn-icon" onclick="resourceManager.downloadResource('${resource.id}')" title="下载">
                        <svg width="16" height="16" viewBox="0 0 24 24" fill="currentColor">
                            <path d="M19 9h-4V3H9v6H5l7 7 7-7zM5 18v2h14v-2H5z"/>
                        </svg>
                    </button>
                    <button class="btn-icon" onclick="resourceManager.deleteResource('${resource.id}')" title="删除" style="color: var(--error);">
                        <svg width="16" height="16" viewBox="0 0 24 24" fill="currentColor">
                            <path d="M6 19c0 1.1.9 2 2 2h8c1.1 0 2-.9 2-2V7H6v12zM19 4h-3.5l-1-1h-5l-1 1H5v2h14V4z"/>
                        </svg>
                    </button>
                </div>
            </div>
        `;
        
        return card;
    }
    
    getFileColor(ext) {
        const colors = {
            pdf: 'linear-gradient(135deg, #FF6B6B, #FF8E8E)',
            docx: 'linear-gradient(135deg, #4ECDC4, #6FE7DD)',
            doc: 'linear-gradient(135deg, #4ECDC4, #6FE7DD)',
            txt: 'linear-gradient(135deg, #95E1D3, #B4EFE3)'
        };
        return colors[ext] || 'linear-gradient(135deg, #BDBDBD, #E0E0E0)';
    }
    
    showEmptyState() {
        this.resourceGrid.innerHTML = `
            <div class="resource-placeholder" style="grid-column: 1 / -1;">
                <svg width="120" height="120" viewBox="0 0 24 24" fill="currentColor" opacity="0.1">
                    <path d="M20 6h-8l-2-2H4c-1.1 0-2 .9-2 2v12c0 1.1.9 2 2 2h16c1.1 0 2-.9 2-2V8c0-1.1-.9-2-2-2z"/>
                </svg>
                <h3>暂无资源</h3>
                <p>上传文件后将在这里显示</p>
                <button onclick="window.location.href='/'" class="btn btn-primary" style="margin-top: 20px;">
                    <svg width="20" height="20" viewBox="0 0 24 24" fill="currentColor" style="margin-right: 8px;">
                        <path d="M9 16h6v-6h4l-7-7-7 7h4zm-4 2h14v2H5z"/>
                    </svg>
                    去上传
                </button>
            </div>
        `;
    }
    
    // === 筛选和排序 ===
    filterResources() {
        const query = this.searchInput?.value.toLowerCase() || '';
        
        this.filteredResources = this.resources.filter(resource => {
            const matchesSearch = !query || 
                resource.filename.toLowerCase().includes(query) ||
                (resource.keywords && resource.keywords.some(kw => kw.toLowerCase().includes(query)));
            
            const matchesDifficulty = this.currentDifficulty === 'all' || 
                resource.difficulty === this.currentDifficulty;
            
            return matchesSearch && matchesDifficulty;
        });
        
        this.sortResources();
    }
    
    sortResources() {
        switch (this.currentSort) {
            case 'name':
                this.filteredResources.sort((a, b) => 
                    a.filename.localeCompare(b.filename));
                break;
            case 'size':
                this.filteredResources.sort((a, b) => 
                    (b.size || 0) - (a.size || 0));
                break;
            case 'date':
            default:
                this.filteredResources.sort((a, b) => 
                    new Date(b.uploaded || 0) - new Date(a.uploaded || 0));
                break;
        }
        
        this.renderResources();
    }
    
    // === 资源操作 ===
    async viewResource(resourceId) {
        const resource = this.resources.find(r => r.id === resourceId);
        if (!resource) return;
        
        const modal = document.getElementById('fileModal');
        const modalTitle = document.getElementById('fileModalTitle');
        const modalBody = document.getElementById('fileModalBody');
        
        if (!modal || !modalTitle || !modalBody) return;
        
        modalTitle.textContent = resource.filename;
        modalBody.innerHTML = `
            <div class="resource-detail">
                <div style="display: grid; gap: 16px; margin-bottom: 24px;">
                    <div class="detail-item">
                        <div class="detail-label">文件类型</div>
                        <div class="detail-value">${resource.filename.split('.').pop().toUpperCase()}</div>
                    </div>
                    <div class="detail-item">
                        <div class="detail-label">文件大小</div>
                        <div class="detail-value">${resource.size ? Utils.formatFileSize(resource.size) : '未知'}</div>
                    </div>
                    <div class="detail-item">
                        <div class="detail-label">上传时间</div>
                        <div class="detail-value">${resource.uploaded ? new Date(resource.uploaded).toLocaleString() : '未知'}</div>
                    </div>
                    ${resource.chunks ? `
                        <div class="detail-item">
                            <div class="detail-label">知识块数量</div>
                            <div class="detail-value">${resource.chunks}</div>
                        </div>
                    ` : ''}
                    ${resource.difficulty ? `
                        <div class="detail-item">
                            <div class="detail-label">难度级别</div>
                            <div class="detail-value">
                                <span class="badge badge-warning">${resource.difficulty}</span>
                            </div>
                        </div>
                    ` : ''}
                </div>
                
                ${resource.summary ? `
                    <div style="margin-bottom: 24px;">
                        <h4 style="margin-bottom: 12px; color: var(--gray-700);">内容摘要</h4>
                        <p style="line-height: 1.8; color: var(--gray-600); background: var(--gray-50); padding: 16px; border-radius: 12px;">
                            ${resource.summary}
                        </p>
                    </div>
                ` : ''}
                
                ${resource.keywords && resource.keywords.length > 0 ? `
                    <div style="margin-bottom: 24px;">
                        <h4 style="margin-bottom: 12px; color: var(--gray-700);">关键词</h4>
                        <div style="display: flex; gap: 8px; flex-wrap: wrap;">
                            ${resource.keywords.map(kw => `
                                <span style="padding: 6px 14px; background: var(--cyan-lighter); color: var(--klein-blue); border-radius: 16px; font-size: 0.9rem;">
                                    ${kw}
                                </span>
                            `).join('')}
                        </div>
                    </div>
                ` : ''}
                
                ${resource.outline ? `
                    <div>
                        <h4 style="margin-bottom: 12px; color: var(--gray-700);">文档大纲</h4>
                        <div style="background: var(--gray-50); padding: 16px; border-radius: 12px;">
                            ${this.renderOutline(resource.outline)}
                        </div>
                    </div>
                ` : ''}
            </div>
        `;
        
        ModalManager.open('fileModal');
    }
    
    renderOutline(outline) {
        if (Array.isArray(outline)) {
            return `
                <ul style="list-style: none; padding: 0;">
                    ${outline.map(item => `
                        <li style="padding: 8px 0; border-bottom: 1px solid var(--gray-200);">
                            ${item}
                        </li>
                    `).join('')}
                </ul>
            `;
        }
        return `<p>${outline}</p>`;
    }
    
    async analyzeResource(resourceId) {
        const resource = this.resources.find(r => r.id === resourceId);
        if (!resource) return;
        
        Utils.showToast('正在分析文档...', 'info');
        
        try {
            const response = await fetch('/analyze_document', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ document_id: resourceId })
            });
            
            const data = await response.json();
            
            if (data.status === 'success') {
                this.showAnalysisResults(data.analysis);
            } else {
                throw new Error(data.message);
            }
        } catch (error) {
            console.error('Analysis error:', error);
            Utils.showToast('分析失败', 'error');
        }
    }
    
    showAnalysisResults(analysis) {
        const modal = document.getElementById('fileModal');
        const modalTitle = document.getElementById('fileModalTitle');
        const modalBody = document.getElementById('fileModalBody');
        
        if (!modal || !modalTitle || !modalBody) return;
        
        modalTitle.textContent = '文档分析结果';
        modalBody.innerHTML = `
            <div class="analysis-results">
                ${analysis.readability ? `
                    <div class="card glass-effect" style="margin-bottom: 16px;">
                        <h4 style="margin-bottom: 12px;">可读性分析</h4>
                        <div class="progress-bar">
                            <div class="progress-fill" style="width: ${analysis.readability}%"></div>
                        </div>
                        <p style="margin-top: 8px; color: var(--gray-600);">可读性评分: ${analysis.readability}/100</p>
                    </div>
                ` : ''}
                
                ${analysis.complexity ? `
                    <div class="card glass-effect" style="margin-bottom: 16px;">
                        <h4 style="margin-bottom: 12px;">复杂度</h4>
                        <span class="badge badge-${this.getComplexityBadge(analysis.complexity)}">
                            ${analysis.complexity}
                        </span>
                    </div>
                ` : ''}
                
                ${analysis.topics && analysis.topics.length > 0 ? `
                    <div class="card glass-effect">
                        <h4 style="margin-bottom: 12px;">主要主题</h4>
                        <div style="display: flex; gap: 8px; flex-wrap: wrap;">
                            ${analysis.topics.map(topic => `
                                <span class="badge badge-primary">${topic}</span>
                            `).join('')}
                        </div>
                    </div>
                ` : ''}
            </div>
        `;
        
        ModalManager.open('fileModal');
    }
    
    getComplexityBadge(complexity) {
        const map = {
            '简单': 'success',
            '中等': 'warning',
            '复杂': 'error'
        };
        return map[complexity] || 'primary';
    }
    
    downloadResource(resourceId) {
        const resource = this.resources.find(r => r.id === resourceId);
        if (!resource) return;
        
        // 触发下载
        window.location.href = `/download/${resourceId}`;
        Utils.showToast('开始下载', 'success');
    }
    
    async deleteResource(resourceId) {
        const resource = this.resources.find(r => r.id === resourceId);
        if (!resource) return;
        
        if (!confirm(`确定要删除 "${resource.filename}" 吗？`)) {
            return;
        }
        
        try {
            const response = await fetch(`/delete_document/${resourceId}`, {
                method: 'DELETE'
            });
            
            const data = await response.json();
            
            if (data.status === 'success') {
                this.resources = this.resources.filter(r => r.id !== resourceId);
                this.filterResources();
                this.updateStats();
                Utils.showToast('删除成功', 'success');
            } else {
                throw new Error(data.message);
            }
        } catch (error) {
            console.error('Delete error:', error);
            Utils.showToast('删除失败', 'error');
        }
    }
    
    // === 批量操作 ===
    async batchDownload() {
        if (this.resources.length === 0) {
            Utils.showToast('暂无资源', 'warning');
            return;
        }
        
        Utils.showToast('批量下载功能开发中...', 'info');
        // 实现批量下载逻辑
    }
    
    async analyzeAll() {
        if (this.resources.length === 0) {
            Utils.showToast('暂无资源', 'warning');
            return;
        }
        
        Utils.showToast('正在批量分析...', 'info');
        // 实现批量分析逻辑
    }
    
    async mergeFiles() {
        if (this.resources.length < 2) {
            Utils.showToast('至少需要2个文件才能合并', 'warning');
            return;
        }
        
        Utils.showToast('合并文档功能开发中...', 'info');
        // 实现文件合并逻辑
    }
    
    async clearAll() {
        if (this.resources.length === 0) {
            Utils.showToast('暂无资源', 'warning');
            return;
        }
        
        if (!confirm(`确定要清空所有 ${this.resources.length} 个资源吗？此操作不可恢复！`)) {
            return;
        }
        
        try {
            const response = await fetch('/clear_all_documents', {
                method: 'DELETE'
            });
            
            const data = await response.json();
            
            if (data.status === 'success') {
                this.resources = [];
                this.filterResources();
                this.updateStats();
                Utils.showToast('已清空所有资源', 'success');
            } else {
                throw new Error(data.message);
            }
        } catch (error) {
            console.error('Clear error:', error);
            Utils.showToast('清空失败', 'error');
        }
    }
    
    // === 统计更新 ===
    updateStats() {
        if (this.totalDocs) {
            this.totalDocs.textContent = this.resources.length;
        }
        
        // 按类型统计
        const pdfFiles = this.resources.filter(r => r.filename.toLowerCase().endsWith('.pdf'));
        const docxFiles = this.resources.filter(r => r.filename.toLowerCase().endsWith('.docx'));
        const txtFiles = this.resources.filter(r => r.filename.toLowerCase().endsWith('.txt'));
        
        if (this.pdfCount) {
            this.pdfCount.textContent = `${pdfFiles.length} 个文件`;
        }
        if (this.docxCount) {
            this.docxCount.textContent = `${docxFiles.length} 个文件`;
        }
        if (this.txtCount) {
            this.txtCount.textContent = `${txtFiles.length} 个文件`;
        }
        
        // 分类数
        const categories = new Set(this.resources.map(r => r.filename.split('.').pop()));
        if (this.totalCategories) {
            this.totalCategories.textContent = categories.size;
        }
        
        // 总知识块数
        const totalChunks = this.resources.reduce((sum, r) => sum + (r.chunks || 0), 0);
        if (this.totalChunks) {
            this.totalChunks.textContent = totalChunks;
        }
    }
    
    // === 关键词云 ===
    renderKeywordCloud(keywords) {
        if (!this.keywordCloud || !keywords || keywords.length === 0) {
            if (this.keywordCloud) {
                this.keywordCloud.innerHTML = `
                    <div class="keyword-placeholder">
                        <p>上传文件后显示关键词云</p>
                    </div>
                `;
            }
            return;
        }
        
        this.keywordCloud.innerHTML = '';
        
        // 按频率排序
        const sortedKeywords = keywords.sort((a, b) => (b.weight || 0) - (a.weight || 0)).slice(0, 20);
        
        sortedKeywords.forEach(kw => {
            const span = document.createElement('span');
            span.className = 'keyword-tag';
            span.textContent = kw.word || kw;
            
            // 根据权重设置大小
            const weight = kw.weight || 1;
            const fontSize = 0.8 + (weight / 10) * 0.8; // 0.8rem - 1.6rem
            span.style.fontSize = `${fontSize}rem`;
            
            // 随机颜色
            const colors = ['var(--klein-blue)', 'var(--cyan)', '#4ECDC4', '#95E1D3'];
            span.style.color = colors[Math.floor(Math.random() * colors.length)];
            
            // 点击搜索该关键词
            span.onclick = () => {
                if (this.searchInput) {
                    this.searchInput.value = kw.word || kw;
                    this.filterResources();
                }
            };
            
            this.keywordCloud.appendChild(span);
        });
    }
}

    if (window.resourceManager) {
        resourceManager.mergeFiles();
    }
}

    if (window.resourceManager) {
        resourceManager.clearAll();
    }
}

    ModalManager.close('fileModal');
}
