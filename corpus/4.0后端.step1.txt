# ========== 文档处理 ==========
def extract_text_from_file(file):
    """提取文件文本"""
    filename = file.filename.lower()
    
    if filename.endswith('.pdf'):
        pdf_reader = PyPDF2.PdfReader(file)
        return '\n'.join([page.extract_text() for page in pdf_reader.pages])
    
    elif filename.endswith('.docx'):
        doc = docx.Document(file)
        return '\n'.join([para.text for para in doc.paragraphs])
    
    elif filename.endswith('.txt'):
        return file.read().decode('utf-8')
    
    return ""

def chunk_text(text, chunk_size=300, overlap=50):
    """分块文本"""
    sentences = re.split(r'[。！？\n]', text)
    chunks = []
    current_chunk = ""
    
    for sentence in sentences:
        if len(current_chunk) + len(sentence) < chunk_size:
            current_chunk += sentence + "。"
        else:
            if current_chunk:
                chunks.append(current_chunk.strip())
            current_chunk = sentence + "。"
    
    if current_chunk:
        chunks.append(current_chunk.strip())
    
    return [chunk for chunk in chunks if len(chunk) > 20]

def analyze_difficulty(text):
    """分析难度"""
    avg_length = sum(len(s) for s in text.split('。')) / max(len(text.split('。')), 1)
    if avg_length > 50: return "高级"
    elif avg_length > 30: return "中级"
    return "基础"

def extract_keywords_from_text(text, top_n=10):
    """提取关键词"""
    words = re.findall(r'[\u4e00-\u9fa5]{2,}', text)
    counter = Counter(words)
    return [{"word": word, "freq": freq} for word, freq in counter.most_common(top_n)]

@app.route('/upload_multiple', methods=['POST'])
def upload_multiple():
    """批量上传文件"""
    global knowledge_base, document_index, file_metadata
    
    try:
        files = request.files.getlist('files')
        
        if not files:
            return jsonify({"success": False, "message": "未选择文件"})
        
        uploaded_files = []
        total_chunks = 0
        
        for file in files:
            if file.filename == '':
                continue
            
            file_id = str(uuid.uuid4())[:8]
            text = extract_text_from_file(file)
            
            if not text or len(text) < 50:
                continue
            
            chunks = chunk_text(text)
            keywords = extract_keywords_from_text(text, 5)
            difficulty = analyze_difficulty(text)
            
            # 保存元数据
            file_metadata[file_id] = {
                "filename": file.filename,
                "upload_time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                "chunks": len(chunks),
                "keywords": keywords,
                "difficulty": difficulty,
                "char_count": len(text)
            }
            
            # 添加到知识库
            for chunk in chunks:
                knowledge_base.append({
                    "text": chunk,
                    "file_id": file_id,
                    "filename": file.filename
                })
            
            uploaded_files.append({
                "filename": file.filename,
                "chunks": len(chunks),
                "keywords": [kw["word"] for kw in keywords[:3]],
                "difficulty": difficulty
            })
            
            total_chunks += len(chunks)
        
        if not knowledge_base:
            return jsonify({"success": False, "message": "文件内容无效"})
        
        # 构建FAISS索引
        texts = [item["text"] for item in knowledge_base]
        embeddings = embedding_model.encode(texts)
        
        dimension = embeddings.shape[1]
        document_index = faiss.IndexFlatL2(dimension)
        document_index.add(embeddings.astype('float32'))
        
        return jsonify({
            "success": True,
            "message": f"成功上传 {len(uploaded_files)} 个文件",
            "files": uploaded_files,
            "total_chunks": total_chunks
        })
    
    except Exception as e:
        return jsonify({"success": False, "message": f"上传失败: {str(e)}"})

@app.route('/ask', methods=['POST'])
def ask():
    """智能问答"""
    try:
        data = request.json
        question = data.get('question', '')
        style = data.get('style', 'simple')
        language = data.get('language', 'zh')
        
        if not knowledge_base or document_index is None:
            return jsonify({"success": False, "message": "请先上传文件"})
        
        # 检索相关文档
        query_embedding = embedding_model.encode([question])
        D, I = document_index.search(query_embedding.astype('float32'), k=3)
        
        context_parts = []
        for idx in I[0]:
            if idx < len(knowledge_base):
                item = knowledge_base[idx]
                context_parts.append(f"[{item['filename']}]\n{item['text']}")
        
        context = "\n\n".join(context_parts)
        
        # 风格提示词
        style_prompts = {
            'simple': '请用通俗易懂的语言',
            'professional': '请用专业学术的语言',
            'inspiring': '请用启发式的方式',
            'concise': '请简明扼要地',
            'detailed': '请详细深入地'
        }
        
        # 语言提示词
        language_names = {
            'zh': '中文', 'en': 'English', 'vi': 'Tiếng Việt',
            'th': 'ภาษาไทย', 'id': 'Bahasa Indonesia',
            'ms': 'Bahasa Melayu', 'fil': 'Filipino',
            'my': 'မြန်မာဘာသာ', 'lo': 'ພາສາລາວ', 'km': 'ភាសាខ្មែរ'

@app.route('/build_knowledge_graph', methods=['POST'])
def build_knowledge_graph():
    """构建知识图谱"""
    try:
        if not knowledge_base:
            return jsonify({"success": False, "message": "请先上传文件"})
        
        # 提取所有关键词
        all_keywords = []
        for file_id, meta in file_metadata.items():
            all_keywords.extend([kw["word"] for kw in meta["keywords"]])
        
        keyword_counter = Counter(all_keywords)
        top_keywords = keyword_counter.most_common(20)
        
        # 构建节点和边
        nodes = [{"id": kw, "label": kw, "value": freq} for kw, freq in top_keywords]
        
        # 简单的关联关系（共现）
        edges = []
        for i, (kw1, _) in enumerate(top_keywords[:10]):
            for kw2, _ in top_keywords[i+1:i+3]:
                edges.append({"from": kw1, "to": kw2})
        
        return jsonify({
            "success": True,
            "graph": {
                "nodes": nodes,
                "edges": edges
            }
        })
    
    except Exception as e:
        return jsonify({"success": False, "message": str(e)})

@app.route('/extract_keywords', methods=['POST'])
def extract_keywords():
    """提取关键知识点"""
    try:
        if not file_metadata:
            return jsonify({"success": False, "message": "请先上传文件"})
        
        all_keywords = []
        for meta in file_metadata.values():
            all_keywords.extend(meta["keywords"])
        
        keyword_counter = {}
        for kw in all_keywords:
            word = kw["word"]
            if word in keyword_counter:
                keyword_counter[word]["freq"] += kw["freq"]
            else:
                keyword_counter[word] = {"word": word, "freq": kw["freq"]}
        
        sorted_keywords = sorted(keyword_counter.values(), key=lambda x: x["freq"], reverse=True)[:15]
        
        return jsonify({
            "success": True,
            "keywords": sorted_keywords
        })
    
    except Exception as e:
        return jsonify({"success": False, "message": str(e)})
