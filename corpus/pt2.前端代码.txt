    <script>
        let selectedFile = null;
        let queryCount = 0;
        
        // Upload handling
        const fileInput = document.getElementById('fileInput');
        const uploadZone = document.getElementById('uploadZone');
        const uploadBtn = document.getElementById('uploadBtn');
        const uploadStatus = document.getElementById('uploadStatus');
        
        uploadZone.onclick = () => fileInput.click();
        
        fileInput.onchange = (e) => {
            selectedFile = e.target.files[0];
            if (selectedFile) {
                uploadStatus.innerHTML = `<span style="color: var(--klein-blue); font-weight: 600;">✅ 已选择: ${selectedFile.name}</span>`;
                uploadBtn.disabled = false;
            }
        };
        
        uploadBtn.onclick = async () => {
            if (!selectedFile) return;
            
            const formData = new FormData();
            formData.append('file', selectedFile);
            
            uploadBtn.disabled = true;
            uploadBtn.innerHTML = '<span class="loading"></span> 灿灿正在处理...';
            uploadStatus.innerHTML = '⏳ 正在提取知识点，请稍候...';
            
            try {
                const response = await fetch('/upload', {
                    method: 'POST',
                    body: formData
                });
                
                const result = await response.json();
                
                if (result.success) {
                    showNotification(`✅ ${result.message}`, 'success');
                    uploadStatus.innerHTML = `
                        <div style="color: var(--success); font-weight: 600;">✅ 上传成功！</div>
                        <div style="font-size: 0.9em; margin-top: 5px;">
                            📊 难度: ${result.difficulty} | 
                            🔑 关键词: ${result.keywords.slice(0, 3).join(', ')}
                        </div>
                    `;
                    document.getElementById('questionInput').disabled = false;
                    document.getElementById('askBtn').disabled = false;
                    updateStats();
                    selectedFile = null;
                    fileInput.value = '';
                } else {
                    showNotification(`❌ ${result.message}`, 'error');
                    uploadStatus.innerHTML = `<span style="color: var(--error);">❌ ${result.message}</span>`;
                }
            } catch (error) {
                showNotification('❌ 上传失败: ' + error.message, 'error');
                uploadStatus.innerHTML = '<span style="color: var(--error);">❌ 上传失败</span>';
            }
            
            uploadBtn.disabled = false;
            uploadBtn.innerHTML = '开始上传';
        };
        
        // Chat handling
        const questionInput = document.getElementById('questionInput');
        const askBtn = document.getElementById('askBtn');
        const chatContainer = document.getElementById('chatContainer');
        
        askBtn.onclick = async () => {
            const question = questionInput.value.trim();
            if (!question) return;
            
            const style = document.getElementById('styleSelect').value;
            const language = document.getElementById('languageSelect').value;
            
            addMessage(question, 'user');
            questionInput.value = '';
            queryCount++;
            updateQueryCount();
            
            askBtn.disabled = true;
            askBtn.innerHTML = '<span class="loading"></span>';
            
            try {
                const response = await fetch('/ask', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({question, style, language})
                });
                
                const result = await response.json();
                
                if (result.success) {
                    addMessage(result.answer, 'ai');
                } else {
                    addMessage(result.message, 'ai');
                }
            } catch (error) {
                addMessage('抱歉，灿灿遇到了问题: ' + error.message, 'ai');
            }
            
            askBtn.disabled = false;
            askBtn.innerHTML = '📤';
        };
        
        questionInput.onkeypress = (e) => {
            if (e.key === 'Enter') askBtn.click();
        };
        
        function addMessage(text, sender) {
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${sender}`;
            
            const avatar = document.createElement('div');
            avatar.className = 'message-avatar';
            avatar.textContent = sender === 'user' ? '👤' : '✨';
            
            const content = document.createElement('div');
            content.className = 'message-content';
            content.innerHTML = sender === 'user' 
                ? `<strong>您</strong><p>${text}</p>`
                : `<strong>灿灿</strong><p>${text}</p>`;
            
            messageDiv.appendChild(avatar);
            messageDiv.appendChild(content);
            chatContainer.appendChild(messageDiv);
            chatContainer.scrollTop = chatContainer.scrollHeight;
        }

            showModal('🗺️ 正在规划学习路径...', '灿灿正在为您设计最优学习路径...');
            
            try {
                const response = await fetch('/generate_path', {

                    headers: {'Content-Type': 'application/json'}

        async function extractKeywords() {
            showModal('🔑 正在提取关键知识点...', '灿灿正在分析核心概念...');
            
            try {
                const response = await fetch('/extract_keywords', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'}
                });
                
                const result = await response.json();
                
                if (result.success) {
                    const keywordsList = result.keywords
                        .map(kw => `• ${kw.word} (出现${kw.freq}次)`)
                        .join('\n');
                    showModal('🔑 关键知识点', keywordsList);
                } else {
                    showModal('❌ 提取失败', result.message);
                }
            } catch (error) {
                showModal('❌ 错误', '提取失败: ' + error.message);
            }
        }

        }

            const notification = document.createElement('div');
            notification.className = `notification ${type}`;
            notification.textContent = message;
            document.body.appendChild(notification);
            
            setTimeout(() => notification.remove(), 4000);
        }

                document.getElementById('fileCount').textContent = stats.total_files;

                console.error('获取统计信息失败:', error);
            }
        }
        
        function updateQueryCount() {
            document.getElementById('queryCount').textContent = queryCount;
        }

        }
        
        // Close modal on outside click
        document.getElementById('resultModal').onclick = (e) => {
            if (e.target.id === 'resultModal') closeModal();
        };
        
        // Initialize
        updateStats();
