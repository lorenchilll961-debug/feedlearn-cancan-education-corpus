"""
============================================
学习路径生成器模块
根据文档内容生成个性化学习路径
============================================
"""

import re
from typing import List, Dict
import logging

logger = logging.getLogger(__name__)


class LearningPathGenerator:
    """学习路径生成器"""
    
    def __init__(self, llm_client):
        """
        初始化
        
        Args:
            llm_client: LLM客户端
        """
        self.llm_client = llm_client
    
    def generate_path(self, content: str, user_level: str = "beginner") -> Dict:
        """
        生成学习路径
        
        Args:
            content: 文档内容
            user_level: 用户水平 (beginner, intermediate, advanced)
            
        Returns:
            学习路径数据
        """
        try:
            logger.info(f"为 {user_level} 级别生成学习路径...")
            
            # 1. 分析内容结构
            structure = self._analyze_content_structure(content)
            
            # 2. 提取主要主题
            topics = self._extract_main_topics(content)
            
            # 3. 生成学习阶段
            stages = self._generate_learning_stages(topics, user_level, structure)
            
            # 4. 估算学习时间
            total_time = sum(stage['estimated_minutes'] for stage in stages)
            
            path = {
                'user_level': user_level,
                'total_stages': len(stages),
                'total_time_minutes': total_time,
                'stages': stages,
                'recommendations': self._generate_recommendations(user_level)
            }
            
            logger.info(f"学习路径生成完成: {len(stages)} 个阶段")
            return path
            
        except Exception as e:
            logger.error(f"生成学习路径失败: {e}")
            return self._generate_fallback_path()
    
    def _analyze_content_structure(self, content: str) -> Dict:
        """分析内容结构"""
        # 统计基本信息
        words = content.split()
        sentences = [s for s in re.split(r'[。！？.!?]', content) if s.strip()]
        paragraphs = [p for p in content.split('\n') if p.strip()]
        
        # 检测标题（简单版）
        headings = re.findall(r'^[一二三四五六七八九十\d]+[、\.．].*$', content, re.MULTILINE)
        
        return {
            'word_count': len(words),
            'sentence_count': len(sentences),
            'paragraph_count': len(paragraphs),
            'heading_count': len(headings),
            'headings': headings[:10]  # 前10个标题
        }
    
    def _extract_main_topics(self, content: str) -> List[str]:
        """提取主要主题"""
        import jieba.analyse
        
        # 提取关键词作为主题
        keywords = jieba.analyse.extract_tags(content, topK=10, withWeight=False)
        
        # 按重要性分组
        topics = []
        for i, kw in enumerate(keywords):
            if i < 3:
                topics.append(f"核心概念：{kw}")
            elif i < 7:
                topics.append(f"重要知识点：{kw}")
            else:
                topics.append(f"扩展内容：{kw}")
        
        return topics
    
    def _generate_learning_stages(self, topics: List[str], 
                                  user_level: str, 
                                  structure: Dict) -> List[Dict]:
        """生成学习阶段"""
        stages = []
        
        # 根据用户水平调整阶段数
        stage_count = {
            'beginner': 5,
            'intermediate': 4,
            'advanced': 3
        }.get(user_level, 4)
        
        # 分配主题到阶段
        topics_per_stage = max(1, len(topics) // stage_count)
        
        for i in range(stage_count):
            start_idx = i * topics_per_stage
            end_idx = start_idx + topics_per_stage if i < stage_count - 1 else len(topics)
            
            stage_topics = topics[start_idx:end_idx]
            
            stage = {
                'stage_number': i + 1,
                'title': self._generate_stage_title(i, stage_count, user_level),
                'description': self._generate_stage_description(i, stage_count, user_level),
                'topics': stage_topics,
                'estimated_minutes': self._estimate_time(len(stage_topics), user_level),
                'difficulty': self._get_stage_difficulty(i, stage_count),
                'objectives': self._generate_objectives(stage_topics),
                'completed': False
            }
            
            stages.append(stage)
        
        return stages
    
    def _generate_stage_title(self, index: int, total: int, level: str) -> str:
        """生成阶段标题"""
        titles = {
            'beginner': [
                "基础入门",
                "核心概念理解",
                "实践应用初步",
                "知识巩固",
                "综合提升"
            ],
            'intermediate': [
                "知识回顾与深化",
                "核心技能掌握",
                "实战演练",
                "高级应用"
            ],
            'advanced': [
                "深度理解与分析",
                "专业技能精进",
                "综合应用与创新"
            ]
        }
        
        level_titles = titles.get(level, titles['intermediate'])
        return level_titles[min(index, len(level_titles) - 1)]
    
    def _generate_stage_description(self, index: int, total: int, level: str) -> str:
        """生成阶段描述"""
        descriptions = {
            0: "建立基础知识框架，了解核心概念和基本原理。",
            1: "深入理解关键知识点，掌握核心技能。",
            2: "通过实践练习巩固所学内容。",
            3: "综合运用所学知识，提升实际应用能力。",
            4: "完善知识体系，达到学习目标。"
        }
        return descriptions.get(index, "继续学习和提升相关技能。")
    
    def _estimate_time(self, topic_count: int, level: str) -> int:
        """估算学习时间（分钟）"""
        base_time = {
            'beginner': 30,
            'intermediate': 25,
            'advanced': 20
        }.get(level, 25)
        
        return base_time * topic_count
    
    def _get_stage_difficulty(self, index: int, total: int) -> str:
        """获取阶段难度"""
        progress = (index + 1) / total
        
        if progress <= 0.3:
            return "简单"
        elif progress <= 0.7:
            return "中等"
        else:
            return "困难"
    
    def _generate_objectives(self, topics: List[str]) -> List[str]:
        """生成学习目标"""
        objectives = []
        
        for topic in topics[:3]:  # 最多3个目标
            objectives.append(f"理解并掌握{topic}")
        
        return objectives
    
    def _generate_recommendations(self, user_level: str) -> List[str]:
        """生成学习建议"""
        recommendations = {
            'beginner': [
                "建议每天学习30-60分钟，保持连续性",
                "遇到难点及时记录，寻求帮助",
                "多做练习题巩固知识点",
                "建议按顺序完成各个阶段"
            ],
            'intermediate': [
                "每天学习45-90分钟效果最佳",
                "注重理论与实践结合",
                "尝试独立解决问题",
                "可以跳过已掌握的内容"
            ],
            'advanced': [
                "建议进行深度学习和研究",
                "注重知识的系统性和完整性",
                "多思考实际应用场景",
                "可以根据需求调整学习顺序"
            ]
        }
        
        return recommendations.get(user_level, recommendations['intermediate'])
    
    def _generate_fallback_path(self) -> Dict:
        """生成备用学习路径"""
        return {
            'user_level': 'beginner',
            'total_stages': 3,
            'total_time_minutes': 120,
            'stages': [
                {
                    'stage_number': 1,
                    'title': "基础入门",
                    'description': "了解基本概念和原理",
                    'topics': ["核心概念", "基础知识"],
                    'estimated_minutes': 40,
                    'difficulty': "简单",
                    'objectives': ["理解核心概念"],
                    'completed': False
                },
                {
                    'stage_number': 2,
                    'title': "知识深化",
                    'description': "深入学习关键内容",
                    'topics': ["重要知识点", "实践应用"],
                    'estimated_minutes': 50,
                    'difficulty': "中等",
                    'objectives': ["掌握关键技能"],
                    'completed': False
                },
                {
                    'stage_number': 3,
                    'title': "综合提升",
                    'description': "综合运用所学知识",
                    'topics': ["综合应用", "高级技巧"],
                    'estimated_minutes': 30,
                    'difficulty': "中等",
                    'objectives': ["达到学习目标"],
                    'completed': False
                }
            ],
            'recommendations': [
                "建议按顺序学习",
                "多做练习巩固知识",
                "遇到问题及时寻求帮助"
            ]
        }
