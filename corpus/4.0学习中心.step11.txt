/* ============================================
   灿灿AI - 学习中心功能
   ============================================ */

class LearningCenterManager {
    constructor() {
        this.currentTest = null;
        this.testResults = [];
        this.learningPath = null;
        this.studyTime = 0;
        this.timer = null;
        
        this.initElements();
        this.initEventListeners();
        this.loadLearningData();
        this.startTimer();
    }
    
    initElements() {
        // 测试相关
        this.generateQuizBtn = document.getElementById('generateQuiz');
        this.quizContainer = document.getElementById('quizContainer');
        this.submitAnswerBtn = document.getElementById('submitAnswer');
        
        // 学习路径
        this.generatePathBtn = document.getElementById('generatePath');
        this.pathContainer = document.getElementById('pathContainer');
        
        // 统计
        this.studyTimeEl = document.getElementById('studyTime');
        this.completionEl = document.getElementById('completion');
        this.accuracyEl = document.getElementById('accuracy');
        
        // 图表
        this.progressChart = document.getElementById('progressChart');
        this.performanceChart = document.getElementById('performanceChart');
    }
    
    initEventListeners() {
        if (this.generateQuizBtn) {
            this.generateQuizBtn.addEventListener('click', () => this.generateQuiz());
        }
        
        if (this.submitAnswerBtn) {
            this.submitAnswerBtn.addEventListener('click', () => this.submitAnswer());
        }
        
        if (this.generatePathBtn) {
            this.generatePathBtn.addEventListener('click', () => this.generateLearningPath());
        }
    }
    
    // === 测试生成 ===
    async generateQuiz() {
        const count = document.getElementById('questionCount')?.value || 5;
        const difficulty = document.getElementById('quizDifficulty')?.value || 'medium';
        const type = document.getElementById('quizType')?.value || 'mixed';
        
        LoadingManager.show(this.quizContainer, '正在生成题目...');
        this.generateQuizBtn.disabled = true;
        
        try {
            const response = await fetch('/generate_questions', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ count, difficulty, type })
            });
            
            const data = await response.json();
            
            if (data.status === 'success') {
                this.currentTest = {
                    questions: data.questions,
                    startTime: Date.now(),
                    answers: []
                };
                this.renderQuiz();
                Utils.showToast('题目生成成功', 'success');
            } else {
                throw new Error(data.message);
            }
        } catch (error) {
            console.error('Quiz generation error:', error);
            Utils.showToast('生成失败', 'error');
        } finally {
            LoadingManager.hide(this.quizContainer);
            this.generateQuizBtn.disabled = false;
        }
    }
    
    renderQuiz() {
        if (!this.currentTest || !this.currentTest.questions) return;
        
        this.quizContainer.innerHTML = '';
        
        this.currentTest.questions.forEach((q, index) => {
            const questionEl = document.createElement('div');
            questionEl.className = 'card glass-effect';
            questionEl.style.marginBottom = '24px';
            
            questionEl.innerHTML = `
                <div style="padding: 24px;">
                    <div style="display: flex; align-items: center; margin-bottom: 16px;">
                        <span class="badge badge-primary">第 ${index + 1} 题</span>
                        <span class="badge badge-warning" style="margin-left: 8px;">${this.getDifficultyText(q.difficulty)}</span>
                    </div>
                    <h3 style="margin-bottom: 20px; font-size: 1.2rem; line-height: 1.6;">
                        ${q.question}
                    </h3>
                    ${this.renderQuestionOptions(q, index)}
                </div>
            `;
            
            this.quizContainer.appendChild(questionEl);
        });
        
        // 添加提交按钮
        const submitBtn = document.createElement('button');
        submitBtn.className = 'btn btn-primary';
        submitBtn.style.width = '100%';
        submitBtn.innerHTML = `
            <svg width="20" height="20" viewBox="0 0 24 24" fill="currentColor" style="margin-right: 8px;">
                <path d="M9 16.17L4.83 12l-1.42 1.41L9 19 21 7l-1.41-1.41z"/>
            </svg>
            提交答案
        `;
        submitBtn.onclick = () => this.submitAnswer();
        
        this.quizContainer.appendChild(submitBtn);
    }
    
    renderQuestionOptions(question, index) {
        if (question.type === 'multiple_choice') {
            return `
                <div class="quiz-options">
                    ${question.options.map((opt, i) => `
                        <label class="quiz-option">
                            <input type="radio" name="q${index}" value="${i}">
                            <span>${String.fromCharCode(65 + i)}. ${opt}</span>
                        </label>
                    `).join('')}
                </div>
            `;
        } else if (question.type === 'true_false') {
            return `
                <div class="quiz-options">
                    <label class="quiz-option">
                        <input type="radio" name="q${index}" value="true">
                        <span>正确</span>
                    </label>
                    <label class="quiz-option">
                        <input type="radio" name="q${index}" value="false">
                        <span>错误</span>
                    </label>
                </div>
            `;
        } else {
            return `
                <textarea 
                    id="answer${index}" 
                    placeholder="请输入你的答案..." 
                    style="width: 100%; min-height: 100px; padding: 12px; border: 2px solid var(--gray-200); border-radius: 8px; font-family: inherit; font-size: 1rem;"
                ></textarea>
            `;
        }
    }
    
    getDifficultyText(difficulty) {
        const map = {
            'easy': '简单',
            'medium': '中等',
            'hard': '困难'
        };
        return map[difficulty] || '中等';
    }
    
    // === 提交答案 ===
    async submitAnswer() {
        if (!this.currentTest) return;
        
        const answers = [];
        
        this.currentTest.questions.forEach((q, index) => {
            if (q.type === 'short_answer') {
                const textarea = document.getElementById(`answer${index}`);
                answers.push(textarea ? textarea.value : '');
            } else {
                const selected = document.querySelector(`input[name="q${index}"]:checked`);
                answers.push(selected ? selected.value : null);
            }
        });
        
        // 检查是否全部回答
        if (answers.some(a => a === null || a === '')) {
            Utils.showToast('请回答所有问题', 'warning');
            return;
        }
        
        LoadingManager.show(this.quizContainer, '正在批改...');
        
        try {
            const response = await fetch('/submit_answers', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    questions: this.currentTest.questions,
                    answers: answers
                })
            });
            
            const data = await response.json();
            
            if (data.status === 'success') {
                this.showResults(data.results);
                this.saveTestResult(data.results);
            } else {
                throw new Error(data.message);
            }
        } catch (error) {
            console.error('Submit error:', error);
            Utils.showToast('提交失败', 'error');
        } finally {
            LoadingManager.hide(this.quizContainer);
        }
    }
    
    showResults(results) {
        const correct = results.filter(r => r.correct).length;
        const total = results.length;
        const score = Math.round((correct / total) * 100);
        
        this.quizContainer.innerHTML = `
            <div class="card glass-effect" style="text-align: center; padding: 40px;">
                <svg width="100" height="100" viewBox="0 0 24 24" fill="currentColor" style="color: ${score >= 60 ? 'var(--success)' : 'var(--error)'}; margin-bottom: 20px;">
                    <path d="M12 2C6.48 2 2 6.48 2 12s4.48 10 10 10 10-4.48 10-10S17.52 2 12 2zm-2 15l-5-5 1.41-1.41L10 14.17l7.59-7.59L19 8l-9 9z"/>
                </svg>
                <h2 style="font-size: 2.5rem; margin-bottom: 10px; color: ${score >= 60 ? 'var(--success)' : 'var(--error)'};">
                    ${score}分
                </h2>
                <p style="font-size: 1.2rem; color: var(--gray-600); margin-bottom: 30px;">
                    正确 ${correct} / ${total} 题
                </p>
                <div style="display: flex; gap: 16px; justify-content: center;">
                    <button class="btn btn-primary" onclick="lcManager.generateQuiz()">
                        再来一次
                    </button>
                    <button class="btn btn-secondary" onclick="lcManager.viewDetailedResults()">
                        查看详情
                    </button>
                </div>
            </div>
        `;
        
        this.currentTestResults = results;
        this.updateStats();
    }
    
    viewDetailedResults() {
        if (!this.currentTestResults) return;
        
        this.quizContainer.innerHTML = '';
        
        this.currentTestResults.forEach((result, index) => {
            const card = document.createElement('div');
            card.className = 'card glass-effect';
            card.style.marginBottom = '20px';
            card.style.borderLeft = `4px solid ${result.correct ? 'var(--success)' : 'var(--error)'}`;
            
            card.innerHTML = `
                <div style="padding: 24px;">
                    <div style="display: flex; align-items: center; margin-bottom: 16px;">
                        <span class="badge ${result.correct ? 'badge-success' : 'badge-error'}">
                            ${result.correct ? '✓ 正确' : '✕ 错误'}
                        </span>
                    </div>
                    <h3 style="margin-bottom: 16px; font-size: 1.1rem;">
                        ${index + 1}. ${this.currentTest.questions[index].question}
                    </h3>
                    <div style="background: var(--gray-50); padding: 16px; border-radius: 8px; margin-bottom: 12px;">
                        <strong>你的答案：</strong> ${result.userAnswer}
                    </div>
                    ${!result.correct ? `
                        <div style="background: rgba(76, 175, 80, 0.1); padding: 16px; border-radius: 8px; margin-bottom: 12px;">
                            <strong>正确答案：</strong> ${result.correctAnswer}
                        </div>
                    ` : ''}
                    ${result.explanation ? `
                        <div style="padding: 16px; background: var(--cyan-lighter); border-radius: 8px;">
                            <strong>解析：</strong> ${result.explanation}
                        </div>
                    ` : ''}
                </div>
            `;
            
            this.quizContainer.appendChild(card);
        });
    }
    
    // === 学习路径 ===
    async generateLearningPath() {
        LoadingManager.show(this.pathContainer, '正在生成学习路径...');
        this.generatePathBtn.disabled = true;
        
        try {
            const response = await fetch('/generate_learning_path', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' }
            });
            
            const data = await response.json();
            
            if (data.status === 'success') {
                this.learningPath = data.path;
                this.renderLearningPath();
                Utils.showToast('学习路径生成成功', 'success');
            } else {
                throw new Error(data.message);
            }
        } catch (error) {
            console.error('Path generation error:', error);
            Utils.showToast('生成失败', 'error');
        } finally {
            LoadingManager.hide(this.pathContainer);
            this.generatePathBtn.disabled = false;
        }
    }
    
    renderLearningPath() {
        if (!this.learningPath) return;
        
        this.pathContainer.innerHTML = `
            <div class="learning-path">
                ${this.learningPath.stages.map((stage, index) => `
                    <div class="path-stage ${stage.completed ? 'completed' : ''}">
                        <div class="path-stage-number">${index + 1}</div>
                        <div class="path-stage-content">
                            <h3>${stage.title}</h3>
                            <p>${stage.description}</p>
                            <div style="margin-top: 12px; display: flex; gap: 8px; flex-wrap: wrap;">
                                ${stage.topics.map(topic => `
                                    <span class="badge badge-primary">${topic}</span>
                                `).join('')}
                            </div>
                            <div style="margin-top: 16px;">
                                <span style="font-size: 0.9rem; color: var(--gray-600);">
                                    预计时间: ${stage.estimatedTime}
                                </span>
                            </div>
                            ${!stage.completed ? `
                                <button class="btn btn-primary btn-small" style="margin-top: 12px;" onclick="lcManager.startStage(${index})">
                                    开始学习
                                </button>
                            ` : ''}
                        </div>
                    </div>
                `).join('')}
            </div>
        `;
    }
    
    startStage(index) {
        Utils.showToast(`开始学习阶段 ${index + 1}`, 'info');
        // 可以实现具体的学习功能
    }
    
    // === 数据持久化 ===
    saveTestResult(results) {
        const correct = results.filter(r => r.correct).length;
        const total = results.length;
        
        const result = {
            date: new Date().toISOString(),
            score: Math.round((correct / total) * 100),
            correct: correct,
            total: total
        };
        
        this.testResults.push(result);
        StorageManager.set('testResults', this.testResults);
        
        this.updateCharts();
    }
    
    loadLearningData() {
        this.testResults = StorageManager.get('testResults', []);
        this.studyTime = StorageManager.get('studyTime', 0);
        
        this.updateStats();
        this.updateCharts();
    }
    
    // === 计时器 ===
    startTimer() {
        this.timer = setInterval(() => {
            this.studyTime++;
            StorageManager.set('studyTime', this.studyTime);
            this.updateTimeDisplay();
        }, 60000); // 每分钟更新
    }
    
    updateTimeDisplay() {
        if (this.studyTimeEl) {
            const hours = Math.floor(this.studyTime / 60);
            const minutes = this.studyTime % 60;
            this.studyTimeEl.textContent = `${hours}h ${minutes}m`;
        }
    }
    
    // === 统计更新 ===
    updateStats() {
        this.updateTimeDisplay();
        
        if (this.testResults.length > 0) {
            const avgScore = this.testResults.reduce((sum, r) => sum + r.score, 0) / this.testResults.length;
            
            if (this.accuracyEl) {
                this.accuracyEl.textContent = Math.round(avgScore) + '%';
            }
            
            if (this.completionEl) {
                this.completionEl.textContent = this.testResults.length;
            }
        }
    }
    
    // === 图表渲染 ===
    updateCharts() {
        this.renderProgressChart();
        this.renderPerformanceChart();
    }
    
    renderProgressChart() {
        // 简化版进度图表（使用CSS）
        if (this.progressChart && this.testResults.length > 0) {
            const recent = this.testResults.slice(-7);
            
            this.progressChart.innerHTML = `
                <div style="display: flex; align-items: flex-end; justify-content: space-around; height: 200px; gap: 8px;">
                    ${recent.map(r => `
                        <div style="flex: 1; display: flex; flex-direction: column; align-items: center;">
                            <div style="width: 100%; height: ${r.score}%; background: var(--gradient-primary); border-radius: 8px 8px 0 0; min-height: 4px;"></div>
                            <span style="font-size: 0.8rem; margin-top: 8px; color: var(--gray-600);">${r.score}%</span>
                        </div>
                    `).join('')}
                </div>
            `;
        }
    }
    
    renderPerformanceChart() {
        // 简化版性能图表
        if (this.performanceChart && this.testResults.length > 0) {
            const avgScore = this.testResults.reduce((sum, r) => sum + r.score, 0) / this.testResults.length;
            
            this.performanceChart.innerHTML = `
                <div style="text-align: center; padding: 40px;">
                    <div style="position: relative; width: 150px; height: 150px; margin: 0 auto;">
                        <svg viewBox="0 0 36 36" style="transform: rotate(-90deg);">
                            <path d="M18 2.0845 a 15.9155 15.9155 0 0 1 0 31.831 a 15.9155 15.9155 0 0 1 0 -31.831" 
                                fill="none" stroke="var(--gray-200)" stroke-width="3"/>
                            <path d="M18 2.0845 a 15.9155 15.9155 0 0 1 0 31.831 a 15.9155 15.9155 0 0 1 0 -31.831" 
                                fill="none" stroke="url(#gradient)" stroke-width="3"
                                stroke-dasharray="${avgScore}, 100"/>
                            <defs>
                                <linearGradient id="gradient">
                                    <stop offset="0%" stop-color="var(--klein-blue)"/>
                                    <stop offset="100%" stop-color="var(--cyan)"/>
                                </linearGradient>
                            </defs>
                        </svg>
                        <div style="position: absolute; top: 50%; left: 50%; transform: translate(-50%, -50%);">
                            <div style="font-size: 2rem; font-weight: 800; color: var(--klein-blue);">
                                ${Math.round(avgScore)}%
                            </div>
                            <div style="font-size: 0.8rem; color: var(--gray-600);">平均分</div>
                        </div>
                    </div>
                </div>
            `;
        }
    }
}
